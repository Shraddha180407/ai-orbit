# LangGraph Autonomous Agent with Tool Protocol and Checkpointing
import operator
from typing import Annotated, Sequence, TypedDict
from langchain_core.messages import BaseMessage, ToolMessage
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.sqlite import SqliteSaver

class AgentState(TypedDict):
    messages: Annotated[Sequence[BaseMessage], operator.add]
    retry_count: int

# Instantiate Sqlite Saver for persistent checkpointing
memory = SqliteSaver.from_conn_string(":memory:")

workflow = StateGraph(AgentState)
workflow.add_node("agent", call_reasoning_model)
workflow.add_node("tools", mcp_tool_executor)
workflow.set_entry_point("agent")

# Dynamic branch on tool call status
workflow.add_conditional_edges(
    "agent",
    evaluate_tool_calls,
    {"continue": "tools", "end": END}
)
app = workflow.compile(checkpointer=memory)
print("✓ LangGraph runtime compiled with SQLite rollback hooks")
